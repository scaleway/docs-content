<Accordion>
  ## Account

  ### Account supported endpoints

    The following [Account](/account/) endpoints are supported and will be logged in Audit Trail:

    - SendPhoneValidation (`POST /account/v3/phone-validations`)
    - ConfirmPhoneValidation (`POST /account/v3/phone-validations/{phone_validation_id}/confirm`)
    - UpdateUserPreferences (`PATCH /account/v3/users/{user_id}/preferences`)
    - CreateMFAOTP (`POST /account/v3/mfa/otps`)
    - DeleteMFAOTP (`DELETE /account/v3/mfa/otps/{mfa_otp_id}`)
    - ValidateMFAOTP (`POST /account/v3/mfa/otps/{mfa_otp_id}/validate`)
    - UpdateUser (`PATCH /account/v3/users/{user_id}`)
    - ChangeUserEmail (`POST /account/v3/users/{user_id}/change-email`)
    - ChangeUserPassword (`POST /account/v3/users/{user_id}/change-password`)
    - SetMailingListSubscriptions (`PUT /account/v3/users/{user_id}/mailing-list-subscriptions`)
    - StartWebAuthnRegistration (`POST /account/v3/start-webauthn-registration`)
    - FinishWebAuthnRegistration (`POST /account/v3/finish-webauthn-registration`)
    - UpdateWebAuthnAuthenticator (`POST /account/v3/update-webauthn-authenticator`)
    - DeleteWebAuthnAuthenticator (`POST /account/v3/delete-webauthn-authenticator`)    
    - ChangeSupportPlan (`POST /account/v3/organizations/{organization_id}/support-plans`)
    - CloseOrganization (`POST /account/v3/organizations/{organization_id}/close`)
    - SetPicture (`PUT /account/v3/organizations/{organization_id}/picture`)
    - SetOrganizationUseCases (`PATCH /account/v3/organization-use-cases`)
    - UpdateOrganizationDetails (`POST /account/v3/organizations/{organization_id}/update-details`)
    - UpdateOrganizationBillingDetails (`POST /account/v3/organizations/{organization_id}/update-billing-details`)    
    - DownloadContractSignature (`GET account/v3/contract-signatures/{contract_signature_id}/download`)
    - CreateContractSignature (`POST /account/v3/contract-signatures`)
    - ValidateContractSignature (`POST /account/v3/contract-signatures/check`)
    - ValidateAccount (`POST /account/v3/validate-account`)
    - SendResetPasswordEmail (`POST /account/v3/send-reset-password-email`)
    - ResetPassword (`POST /account/v3/reset-password`)
    - InitiatePasswordlessLogin (`POST /account/v3/initiate-passwordless-login`)
    - InitiateAuthenticationCodeLogin (`POST /account/v3/initiate-authentication-code-login`)
    - ResendAuthenticationCode (`POST /account/v3/resend-authentication-code`)
    - CreateLoginSession (`POST /account/v3/login-session`)
    - LogIn (`POST /account/v3/login`)
    - LogInOrganizations (`POST /account/v3/login-organizations`)
    - ValidateEmailUpdate (`POST /account/v3/users/{account_root_user_id}/validate-email-update`)
    - RejectEmailUpdate (`POST /account/v3/users/{account_root_user_id}/reject-email-update`)
    - StartWebAuthnAuthentication (`POST /account/v3/start-webauthn-authentication`)
    - CreateProject (`POST /account/v3/projects`)
    - DeleteProject (`DELETE /account/v3/projects/{project_id}`)
    - UpdateProject (`PATCH /account/v3/projects/{project_id}`)
    - SetProjectQualification (`POST /account/v3/projects/{project_id}/project-qualification`)
    - CreateGDPRRequest (`POST /account/v2/gdpr-requests`)
    - DownloadGDPRExport (`GET /account/v2/gdpr-exports/{filename}/download`)
</Accordion>
